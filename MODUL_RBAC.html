Struktur Tabel RBAC (Spatie)

TabelFungsi
rolesDaftar role (admin, mahasiswa, dosen, ...).
permissionsDaftar permission: CRUD (lihat-data, ...) dan menu (menu-*).
model_has_rolesRelasi user ↔ role.
model_has_permissionsRelasi user ↔ permission (langsung).
role_has_permissionsRelasi role ↔ permission — diubah lewat menu RBAC Control.

A. RoleController

File: app/Http/Controllers/RoleController.php BARU

CRUD role sekaligus menetapkan permission ke role (role_has_permissions). Menu RBAC Control punya dua tampilan menu sekaligus: daftar role (dengan permission & jumlah pengguna) dan form tambah/ubah role berisi checkbox permission.

<?php

namespace App\Http\Controllers;

use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;
use Illuminate\View\View;
use Spatie\Permission\Models\Permission;
use Spatie\Permission\Models\Role;
use Spatie\Permission\PermissionRegistrar;

class RoleController extends Controller
{
    public function index(): View
    {
        $roles = Role::withCount('users')->with('permissions')->orderBy('id')->get();

        return view('role.index', compact('roles'));
    }

    public function create(): View
    {
        $permissions = Permission::orderBy('name')->get();

        return view('role.create', compact('permissions'));
    }

    public function store(Request $request): RedirectResponse
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:50', Rule::unique('roles', 'name')],
            'permissions' => ['nullable', 'array'],
            'permissions.*' => ['exists:permissions,name'],
        ]);

        $role = Role::create(['name' => $validated['name']]);
        $role->syncPermissions($validated['permissions'] ?? []);

        app(PermissionRegistrar::class)->forgetCachedPermissions();

        return redirect()
            ->route('role.index')
            ->with('success', 'Data role berhasil ditambahkan.');
    }

    public function edit(Role $role): View
    {
        $permissions = Permission::orderBy('name')->get();

        return view('role.edit', compact('role', 'permissions'));
    }

    public function update(Request $request, Role $role): RedirectResponse
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:50', Rule::unique('roles', 'name')->ignore($role->id)],
            'permissions' => ['nullable', 'array'],
            'permissions.*' => ['exists:permissions,name'],
        ]);

        $role->update(['name' => $validated['name']]);
        $role->syncPermissions($validated['permissions'] ?? []);

        app(PermissionRegistrar::class)->forgetCachedPermissions();

        return redirect()
            ->route('role.index')
            ->with('success', 'Data role berhasil diubah.');
    }

    public function destroy(Role $role): RedirectResponse
    {
        if ($role->users()->exists()) {
            return redirect()
                ->route('role.index')
                ->with('error', 'Role tidak bisa dihapus karena masih dipakai oleh pengguna.');
        }

        $role->delete();

        app(PermissionRegistrar::class)->forgetCachedPermissions();

        return redirect()
            ->route('role.index')
            ->with('success', 'Data role berhasil dihapus.');
    }
}
  • withCount('users') + with('permissions') agar daftar role tidak N+1.
  • syncPermissions() menyinkronkan centang permission; forgetCachedPermissions() mereset cache Spatie.
  • destroy() menolak menghapus role yang masih dipakai pengguna (flash error).

A. UserController

File: app/Http/Controllers/UserController.php BARU

CRUD akun pengguna: nama, email, password, dan satu role (dropdown). Password otomatis di-hash oleh cast 'password' => 'hashed' pada model User.

<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;
use Illuminate\View\View;
use Spatie\Permission\Models\Role;

class UserController extends Controller
{
    public function index(): View
    {
        $users = User::with('roles')->orderBy('id')->get();

        return view('pengguna.index', compact('users'));
    }

    public function create(): View
    {
        $roles = Role::orderBy('name')->get();

        return view('pengguna.create', compact('roles'));
    }

    public function store(Request $request): RedirectResponse
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:100'],
            'email' => ['required', 'email', 'max:100', Rule::unique('users', 'email')],
            'password' => ['required', 'string', 'min:8'],
            'role' => ['required', 'exists:roles,name'],
        ]);

        $user = User::create([
            'name' => $validated['name'],
            'email' => $validated['email'],
            'password' => $validated['password'],
        ]);
        $user->syncRoles([$validated['role']]);

        return redirect()
            ->route('pengguna.index')
            ->with('success', 'Data pengguna berhasil ditambahkan.');
    }

    public function edit(User $pengguna): View
    {
        $roles = Role::orderBy('name')->get();

        return view('pengguna.edit', compact('pengguna', 'roles'));
    }

    public function update(Request $request, User $pengguna): RedirectResponse
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:100'],
            'email' => ['required', 'email', 'max:100', Rule::unique('users', 'email')->ignore($pengguna->id)],
            'password' => ['nullable', 'string', 'min:8'],
            'role' => ['required', 'exists:roles,name'],
        ]);

        $pengguna->name = $validated['name'];
        $pengguna->email = $validated['email'];

        if (! empty($validated['password'])) {
            $pengguna->password = $validated['password'];
        }

        $pengguna->save();
        $pengguna->syncRoles([$validated['role']]);

        return redirect()
            ->route('pengguna.index')
            ->with('success', 'Data pengguna berhasil diubah.');
    }

    public function destroy(User $pengguna): RedirectResponse
    {
        if ($pengguna->is(auth()->user())) {
            return redirect()
                ->route('pengguna.index')
                ->with('error', 'Anda tidak bisa menghapus akun yang sedang digunakan.');
        }

        $pengguna->delete();

        return redirect()
            ->route('pengguna.index')
            ->with('success', 'Data pengguna berhasil dihapus.');
    }
}
  • Password nullable saat update: kosong = tidak diubah.
  • destroy() menolak menghapus akun yang sedang login.

Cara Pakai di UI

  1. Login sebagai admin.
  2. Menu RBAC Control → Tambah Role (mis. dosen) → centang menu/permission yang boleh.
  3. Menu Pengguna → Tambah Pengguna → isi nama/email/password → pilih role.
  4. Login sebagai user tersebut → hanya menu yang diizinkan yang tampil.

B. RolePermissionSeeder

File: database/seeders/RolePermissionSeeder.php BERUBAH

Menambah permission menu dan meng-assign-nya: role admin mendapat semua menu, role mahasiswa mendapat menu-jadwal-kuliah saja (contoh). Idempoten, aman dijalankan ulang.

<?php

namespace Database\Seeders;

use App\Models\User;
use Illuminate\Database\Seeder;
use Spatie\Permission\Models\Permission;
use Spatie\Permission\Models\Role;
use Spatie\Permission\PermissionRegistrar;

class RolePermissionSeeder extends Seeder
{
    /**
     * Jalankan seeder role, permission, dan user default.
     */
    public function run(): void
    {
        // Reset cache permission agar data terbaru langsung terbaca.
        app(PermissionRegistrar::class)->forgetCachedPermissions();

        // 1. Buat permission dasar + permission menu (dipakai untuk RBAC per menu).
        $permissions = ['lihat-data', 'tambah-data', 'ubah-data', 'hapus-data'];

        $menuPermissions = [
            'menu-mahasiswa', 'menu-dosen', 'menu-mata-kuliah',
            'menu-ruangan', 'menu-kelas', 'menu-jadwal-kuliah',
        ];

        foreach (array_merge($permissions, $menuPermissions) as $permission) {
            Permission::firstOrCreate(['name' => $permission]);
        }

        // 2. Buat role dan assigns permission.
        // Role admin: semua permission + semua menu.
        $roleAdmin = Role::firstOrCreate(['name' => 'admin']);
        $roleAdmin->syncPermissions(array_merge($permissions, $menuPermissions));

        // Role mahasiswa: read-only + hanya menu Jadwal Kuliah (contoh, bisa diubah dari UI RBAC Control).
        $roleMahasiswa = Role::firstOrCreate(['name' => 'mahasiswa']);
        $roleMahasiswa->syncPermissions(['lihat-data', 'menu-jadwal-kuliah']);

        // 3. Buat user default (password otomatis di-hash oleh cast 'hashed' pada model User).
        $admin = User::firstOrCreate(
            ['email' => 'admin@example.test'],
            ['name' => 'Administrator', 'password' => 'password']
        );
        $admin->syncRoles(['admin']);

        $mahasiswa = User::firstOrCreate(
            ['email' => 'mahasiswa@example.test'],
            ['name' => 'Mahasiswa Demo', 'password' => 'password']
        );
        $mahasiswa->syncRoles(['mahasiswa']);

        // Reset cache kembali setelah data tersimpan.
        app(PermissionRegistrar::class)->forgetCachedPermissions();
    }
}

B. routes/web.php

File: routes/web.php BERUBAH

Versi final: tiap menu punya grup permission:menu-<slug>. Baca (index/show) cukup permission menu; tulis (create/store/edit/update/destroy) tambahan role:admin. Resource role & pengguna tetap khusus admin.

use Illuminate\Support\Facades\Route;
use App\Http\Controllers\AuthController;
use App\Http\Controllers\DosenController;
use App\Http\Controllers\JadwalKuliahController;
use App\Http\Controllers\KelasController;
use App\Http\Controllers\MahasiswaController;
use App\Http\Controllers\MataKuliahController;
use App\Http\Controllers\RoleController;
use App\Http\Controllers\RuanganController;
use App\Http\Controllers\UserController;

Route::get('/', function () {
    return view('welcome');
});

/*
|--------------------------------------------------------------------------
| Route Autentikasi
|--------------------------------------------------------------------------
| Halaman login hanya untuk tamu (guest). Logout butuh status login.
*/
Route::middleware('guest')->group(function () {
    Route::get('/login', [AuthController::class, 'showLoginForm'])->name('login');
    Route::post('/login', [AuthController::class, 'login']);
});

Route::post('/logout', [AuthController::class, 'logout'])
    ->middleware('auth')
    ->name('logout');

/*
|--------------------------------------------------------------------------
| Route Menu Konten: dijaga permission "menu-<slug>"
|--------------------------------------------------------------------------
| Menu yang boleh tampil diatur per role lewat permission menu (RBAC Control).
| Aksi tulis (create/store/edit/update/destroy) tetap khusus role admin.
| Grup tulis didaftarkan LEBIH DULU agar "create" tidak tertangkap "{id}".
*/
Route::middleware('auth')->group(function () {
    Route::middleware('permission:menu-mahasiswa')->group(function () {
        Route::resource('mahasiswa', MahasiswaController::class)->except(['index', 'show'])->middleware('role:admin');

B. Model User

File: app/Models/User.php BERUBAH

Menambah helper firstAccessibleMenuUrl() untuk menentukan menu pertama yang boleh diakses (redirect setelah login), supaya role tanpa menu-mahasiswa tidak mendarat di 403.

<?php

namespace App\Models;

// use Illuminate\Contracts\Auth\MustVerifyEmail;
use Database\Factories\UserFactory;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use Spatie\Permission\Traits\HasRoles;

class User extends Authenticatable
{
    /** @use HasFactory<UserFactory> */
    use HasFactory, Notifiable, HasRoles;

    /**
     * The attributes that are mass assignable.
     *
     * @var list<string>
     */
    protected $fillable = [
        'name',
        'email',
        'password',
    ];

    /**
     * The attributes that should be hidden for serialization.
     *
     * @var list<string>
     */
    protected $hidden = [
        'password',
        'remember_token',
    ];

    /**
     * Get the attributes that should be cast.
     *
     * @return array<string, string>
     */
    protected function casts(): array
    {
        return [
            'email_verified_at' => 'datetime',
            'password' => 'hashed',
        ];
    }

    /**
     * URL menu pertama yang boleh diakses user (berdasarkan permission menu).
     */
    public function firstAccessibleMenuUrl(): string
    {
        foreach (config('menu') as $menu) {
            if ($this->can($menu['permission'])) {
                return route($menu['route']);
            }
        }

        return '/';
    }
}

B. AuthController

File: app/Http/Controllers/AuthController.php BERUBAH

Setelah login berhasil, redirect ke menu pertama yang boleh diakses.

<?php

namespace App\Http\Controllers;

use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Validation\ValidationException;
use Illuminate\View\View;

class AuthController extends Controller
{
    /**
     * Tampilkan form login.
     */
    public function showLoginForm(): View
    {
        return view('auth.login');
    }

    /**
     * Proses login.
     */
    public function login(Request $request): RedirectResponse
    {
        $credentials = $request->validate([
            'email' => ['required', 'email'],
            'password' => ['required'],
        ]);

        if (Auth::attempt($credentials, $request->boolean('remember'))) {
            $request->session()->regenerate();

            return redirect()->intended(auth()->user()->firstAccessibleMenuUrl());
        }

        throw ValidationException::withMessages([
            'email' => ['Email atau password salah.'],
        ]);
    }

    /**
     * Proses logout.
     */
    public function logout(Request $request): RedirectResponse
    {
        Auth::logout();

        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return redirect()->route('login');
    }
}

Verifikasi & Skenario Uji

Bagian A

  • Route role & pengguna: 6 rute masing-masing, semua auth, role:admin.
  • Tambah/hapus role + centang permission; hapus role yang dipakai user → ditolak.
  • Tambah pengguna + role; ubah tanpa isi password → password lama tetap berlaku; hapus akun sendiri → ditolak.

Bagian B

  • Permission tersimpan: admin = 4 CRUD + 6 menu-*; mahasiswa = lihat-data + menu-jadwal-kuliah.
  • Route mahasiswa: index/show auth, permission:menu-mahasiswa; tulis auth, permission:menu-mahasiswa, role:admin.
  • Admin → nav semua menu; buat role dosen (Mata Kuliah + Jadwal Kuliah) → login dosen hanya 2 menu, /mahasiswa → 403; mahasiswa hanya 1 menu, /dosen → 403; tiap user mendarat di menu pertamanya.
